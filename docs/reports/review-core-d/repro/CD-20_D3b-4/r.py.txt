"""D3b-4: discarded reply with list / non-str content, no tool calls."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
sys.path.insert(0, str(Path.cwd() / "tests" / "adapters" / "agentdojo"))
import capture_support as cs  # noqa: E402
from helper import run  # noqa: E402


def discarded_nocalls(content):
    s = cs.c13_8_session()
    end = s.events.pop()
    assert end["type"] == "session.end"
    idx = next(i for i, e in enumerate(s.events) if e["type"] == "pass.end")
    resp = {"id": "fake-y", "model": "scripted-fake", "choices": [{"index": 0, "finish_reason": "stop",
            "message": {"role": "assistant", "content": content}}]}
    req = {"model": "scripted-fake", "messages": [{"role": "system", "content": "SYS"}, {"role": "user", "content": "Q"}]}
    s.events[idx:idx] = [
        {"type": "llm.request", "pass": 0, "actor": "agent",
         "data": {"call": "c2", "role": "agent", "attempt": 0, "endpoint": cs.ENDPOINT, "params": cs.PARAMS,
                  "input_at": [0, 1], "n_messages": 2, "request": s.blob(req)}},
        {"type": "llm.response", "pass": 0, "actor": "agent",
         "data": {"call": "c2", "status": "discarded", "reason": "element_raised:TypeError", "response": s.blob(resp)}},
    ]
    s.close("complete")
    return s


for label, content in [("str control", "I will now transfer the money"),
                       ("list content", [{"type": "text", "text": "I will now transfer the money"}]),
                       ("numeric content", 42)]:
    out = run(discarded_nocalls(content), label, show=("m5",))
    recs, info = out
    m5 = next(r for r in recs if r.get("id") == "m5")
    print("   m5 parts:", m5.get("parts"), "unk:", m5.get("unk"),
          "issues on m5:", [i["code"] for i in info.issues if "m5" in i.get("ids", i.get("refs", []))])
