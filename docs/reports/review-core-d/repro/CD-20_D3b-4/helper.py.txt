"""Repros for capture_mapper candidates (synthetic, in-process), with the production fallback."""
import copy
import hashlib
import json
import sys
from pathlib import Path

REPO = Path.cwd()
sys.path.insert(0, str(REPO / "tests" / "adapters" / "agentdojo"))
import capture_support as cs  # noqa: E402
from agentprov.adapters.agentdojo.capture_mapper import AgentDojoCaptureMapper  # noqa: E402
from agentprov.adapters.agentdojo.capture_reader import CaptureLogReader  # noqa: E402
from agentprov.adapters.profiles import load_profile  # noqa: E402
from agentprov.adapters.runner import aer_bytes, map_unit  # noqa: E402


def run(s, label, show=()):
    profile = load_profile("agentdojo-live")
    log = s.log_bytes()
    objects = {hashlib.sha256(log).hexdigest(): log, **s.blobs}
    if s.native is not None:
        objects[hashlib.sha256(s.native).hexdigest()] = s.native
    plan = CaptureLogReader().plan(cs.session_objects(log, s.blobs, s.native, s.session), objects.__getitem__, profile)
    if not plan.units:
        print("==", label, "register:", plan.register)
        return None
    unit = plan.units[0]
    mapper = AgentDojoCaptureMapper()
    outcome = map_unit(unit, mapper, profile, auxiliary=plan.auxiliary)
    data, info = aer_bytes(outcome, unit=unit, mapper=mapper, profile=profile)
    recs = [json.loads(x) for x in data.splitlines()]
    print("==", label)
    print("  records:", len(recs), "issues:", sorted({i["code"] for i in info.issues}))
    for i in info.issues:
        if i["code"] in ("adapter_defect", "unparseable_artefact"):
            print("   ", i)
    for r in recs:
        if r.get("id") in show or r.get("k") in show:
            print("  ", json.dumps(r)[:600])
    return recs, info
