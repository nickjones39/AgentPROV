def test_spread_on_nested_archive(tmp_path, monkeypatch):
    T._profile(monkeypatch)
    pin, tree, _inner = T._nested_input(
        tmp_path,
        direct={"logs/d.eval": T._eval(3, samples=2)},
        inner={"logs/x.eval": T._eval(10, samples=2), "logs/y.eval": T._eval(11, samples=2)},
    )
    seen = {}
    orig = importer.select_sample
    def spy(keys, sample):
        seen["keys"] = list(keys)
        out = orig(keys, sample)
        seen["chosen"] = sorted(out)
        return out
    monkeypatch.setattr(importer, "select_sample", spy)
    report = T._import(tree, pin, tmp_path / "c", limits=T.SMALL, sample=importer.ImportSample(3, "spread"))
    for k, s in seen["keys"]:
        print("KEY", k, "STRATUM", s)
    for k in seen["chosen"]:
        print("CHOSEN", k)
    print("mapped", report.counters.get("mapped"))
    logs_chosen = {k.rsplit("!/", 1)[0] for k in seen["chosen"]}
    assert len(logs_chosen) == 3, f"spread over 3 .eval logs chose from only {sorted(logs_chosen)}"
