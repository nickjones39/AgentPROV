"""Cost of the qwencode salvage path on synthetic, fixture-scale inputs."""
import json, sys, time
from agentprov.adapters.qwencode import dialect
from agentprov.adapters.qwencode.dialect import QwenJsonReader, parse_output, SALVAGE_TRIES
from agentprov.ingest.limits import IngestLimits
from agentprov.ingest.loaders import strict_json_loads
from agentprov.errors import IngestError

L = IngestLimits()
calls = {"n": 0}
_orig = dialect.strict_json_loads
def counting(data, limits=None, **kw):
    calls["n"] += 1
    return _orig(data, limits, **kw)
dialect.strict_json_loads = counting

def t(label, data, fn):
    calls["n"] = 0
    t0 = time.perf_counter()
    try:
        r = fn(data)
        out = type(r).__name__
    except Exception as e:  # noqa: BLE001
        out = f"{type(e).__name__}:{getattr(e,'code','')}"
    dt = time.perf_counter() - t0
    print(f"{label}: bytes={len(data)} strict_parses={calls['n']} secs={dt:.2f} -> {out}", flush=True)
    return dt

# A) unsalvageable (invalid byte early), many '}' : salvage tries bounded at 4096 but each re-parses the prefix
print("SALVAGE_TRIES", SALVAGE_TRIES)
for n in (1 << 10, 1 << 12, 1 << 13):
    d = b'[{"a":1}x' + b'{"b":{}}' * n
    t(f"A n={n}", d, lambda x: parse_output(x, L))
# one strict parse of the 1 MiB doc
big = b'[{"a":1}x' + b'{"b":{}}' * (1 << 17) + b']'
t0 = time.perf_counter()
try: _orig(big, L)
except IngestError as e: pass
one = time.perf_counter() - t0
print(f"one strict parse of {len(big)} bytes: {one:.3f}s; x4096 = {one*4096/60:.1f} min")

# B) salvageable truncated output with many top-level messages -> quadratic _spans
init = {"type": "system", "subtype": "init", "qwen_code_version": "0.21.0"}
for m in (250, 500, 1000):
    msgs = [init] + [{"type": "assistant", "message": {"content": [{"t": "x"}]}} for _ in range(m)]
    full = json.dumps(msgs, separators=(",", ":")).encode()
    cut = full[: len(full) - 10]   # cut mid last message
    t(f"B msgs={m}", cut, lambda x: QwenJsonReader().sniff(x, label="", limits=L))
