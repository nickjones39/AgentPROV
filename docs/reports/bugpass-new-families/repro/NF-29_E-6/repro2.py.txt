def find(o):
    if isinstance(o, dict):
        if "snapshot_id" in o: yield o
        for v in o.values(): yield from find(v)
    elif isinstance(o, list):
        for v in o: yield from find(v)
for sn in find(man): print("snapshot context roles:", [x["role"] for x in sn.get("context", [])])
