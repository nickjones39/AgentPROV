def check(m):
    m = copy.deepcopy(m)
    src = m["sources"][0]
    for s in src["snapshots"]: s["pin"] = {"kind": "stub"}  # in-memory only: pass the D-33 pin check
    spec = NS(corpus_version="0.1.0", mode="import",
              source=NS(source_id=src["source_id"], profile_id=src["profile"]["id"], profile_version=src["profile"]["version"],
                        parts=src["parts"], snapshots=[NS(snapshot_id=s["snapshot_id"]) for s in src["snapshots"]]))
    try: pkg._check_working(m, spec, "0.1.0"); return "ACCEPTED"
    except Exception as e: return f"{type(e).__name__}: {e}"
