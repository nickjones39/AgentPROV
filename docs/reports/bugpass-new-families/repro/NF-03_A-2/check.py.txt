def find(o, k, path=""):
    if isinstance(o, dict):
        for kk, v in o.items():
            if kk == k: yield path+"/"+kk, v
            yield from find(v, k, path+"/"+kk)
    elif isinstance(o, list):
        for i, v in enumerate(o): yield from find(v, k, f"{path}/{i}")
