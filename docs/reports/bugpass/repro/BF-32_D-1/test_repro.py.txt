"""D-1 repro: resuming a derive under another --semantics-profile is accepted (should be refused, exit 4)."""
from __future__ import annotations

import json
import sys
from pathlib import Path
from typing import Any

WT = Path("~/Desktop/PhD/Labs/AgentPROV-framework/.claude/worktrees/bugpass-perf-10-review-b5384c")
for p in (WT / "tests/unit/raw", WT / "tests/unit/corpus"):
    sys.path.insert(0, str(p))

import derive_support as ds  # noqa: E402
import pytest  # noqa: E402

from agentprov.cli.main import main  # noqa: E402


@pytest.fixture(autouse=True)
def _home(monkeypatch, tmp_path_factory):
    monkeypatch.setenv("AGENTPROV_HOME", str(tmp_path_factory.mktemp("home")))
    monkeypatch.delenv("AGENTPROV_PROGRESS", raising=False)


def run(capsys, *argv):
    capsys.readouterr()
    code = main([*argv, "--json", "--progress", "off"])
    out, err = capsys.readouterr()
    return code, out, err


def test_resume_under_another_semantics_profile_is_refused(capsys, tmp_path, monkeypatch):
    from agentprov.corpus import derive

    src = ds.agentdojo_corpus(tmp_path, name="aer-only", count=2, until="aer")
    out = tmp_path / "new"
    real = derive.Derivation._finish_unit
    calls = {"n": 0}

    def flaky(self: Any, *a: Any, **k: Any) -> None:
        calls["n"] += 1
        if calls["n"] == 2:
            raise KeyboardInterrupt
        real(self, *a, **k)

    monkeypatch.setattr(derive.Derivation, "_finish_unit", flaky)
    code1, _o, err1 = run(capsys, "derive", "--corpus", str(src), "--out", str(out), "--in-process",
                          "--semantics-profile", "agentdojo.suites@2")
    print("first run exit", code1, "rows", len(ds.rows(ds.partial(out))))
    assert code1 == 130 and len(ds.rows(ds.partial(out))) == 1
    monkeypatch.setattr(derive.Derivation, "_finish_unit", real)
    code2, out2, err2 = run(capsys, "derive", "--corpus", str(src), "--out", str(out), "--in-process",
                            "--semantics-profile", "agentdojo.suites@3")
    sem_dirs = sorted(p.name for p in (ds.partial(out) / "sem").iterdir()) if (ds.partial(out) / "sem").exists() else []
    manifest = json.loads((ds.partial(out) / "corpus.json").read_text())
    print("second run exit", code2, "counters", json.loads(out2.splitlines()[-1]).get("counters") if out2 else None)
    print("sem dirs", sem_dirs, "manifest semantics", manifest.get("profiles", {}).get("semantics"))
    assert code2 == 4, f"resume under another semantics profile accepted: exit {code2}, sem dirs {sem_dirs}"
