"""A2b-4: the adprov.transcript_v1 mapper silently drops / reshapes v1 members (C.15.3 Import direction).

Each case starts from the C.16.2 source transcript (regenerated byte-exactly from the normative example AER) with one
change. Acceptable outcomes: a byte-identical round trip, a degenerate outcome, or at least one seal issue that the
unmodified source does not raise. A silent value change (no new issue, different bytes) fails.
"""

from __future__ import annotations

import copy
import json
from pathlib import Path

import pytest

from agentprov.adapters.runner import aer_bytes, map_artefact
from agentprov.adapters.types import Origin
from agentprov.views.transcript_v1 import render, serialise

REPO = Path(".")
VEC = REPO / "docs/architecture/examples/C/adprov_v1_llama_injecagent.aer.jsonl"
SRC = serialise(render([json.loads(line) for line in VEC.read_bytes().splitlines()]))


def _map(src: bytes):
    origin = Origin.corpus("AgentDojo-PROV", "2.3", "x/y.transcript.json",
                           producer_commit="579a908354b93361efe0905ffb7623256aa96a06")
    outcome, _ = map_artefact(src, origin, profile_id="adprov-v1")
    if outcome.degenerate:
        return outcome, None
    data, _ = aer_bytes(outcome)
    return outcome, serialise(render([json.loads(line) for line in data.splitlines()]))


def _codes(outcome) -> set[str]:
    return {i.get("code") for i in outcome.issues}


def _released() -> dict:
    return json.loads(SRC)


def _converted() -> dict:
    d = json.loads(SRC)
    meta = d["meta"]
    meta.pop("model_config")
    meta["model"] = meta["model_id"] = "pipe"
    meta["started_at"] = meta["ended_at"] = None
    meta["agentdojo_version"] = meta["benchmark_version"] = None
    meta.update({"source": "agentdojo_native_run_log", "pipeline_name": "pipe", "timing_stubbed": True})
    for t in d["turns"]:
        if t["role"] == "tool":
            t["started_at"] = t["ended_at"] = "0"
    return d


def _insert_model_config(d, value):
    """Place model_config in its J.3.2 layout slot (after benchmark_version, before source)."""
    items = list(d["meta"].items())
    i = [k for k, _ in items].index("benchmark_version") + 1
    d["meta"] = dict(items[:i] + [("model_config", value)] + items[i:])


#: Issues that a case triggers for an unrelated reason (usage [1] has no n_llm_calls, so the call count is unknown).
UNRELATED = {"c_usage_not_object": {"earlier_attempts_unobservable"}}


def _mc(d):
    return d["meta"]["model_config"]


CASES = {
    "a_run_error_released": (_released, lambda d: d["meta"].__setitem__("run_error", "x")),
    "a_source_released": (_released, lambda d: d["meta"].__setitem__("source", "x")),
    "a_timing_stubbed_released": (_released, lambda d: d["meta"].__setitem__("timing_stubbed", False)),
    "a_pipeline_name_released": (_released, lambda d: d["meta"].__setitem__("pipeline_name", "p")),
    "b_finish_reasons_null": (_released, lambda d: _mc(d).__setitem__("finish_reasons", None)),
    "c_usage_not_object": (_released, lambda d: _mc(d).__setitem__("usage", [1])),
    "d_converted_temperature": (_converted, lambda d: _insert_model_config(d, {"temperature": 0.0})),
}


@pytest.mark.parametrize("name", sorted(CASES))
def test_no_silent_value_change(name: str) -> None:
    base_fn, mutate = CASES[name]
    base = base_fn()
    base_src = json.dumps(base, indent=2).encode()
    base_outcome, base_view = _map(base_src)
    assert base_view == base_src, "unmodified base must round-trip"
    d = copy.deepcopy(base)
    mutate(d)
    src = json.dumps(d, indent=2).encode()
    outcome, view = _map(src)
    if outcome.degenerate or view == src:
        return
    new_codes = _codes(outcome) - _codes(base_outcome) - UNRELATED.get(name, set())
    assert new_codes, f"{name}: view bytes differ from source and no new seal issue was raised"
