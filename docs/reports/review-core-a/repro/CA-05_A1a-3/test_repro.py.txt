"""A1a-3: a scalar ``artifacts`` in an untrusted header must yield findings, not a TypeError, when a store is given."""
import sys
from pathlib import Path

import pytest

_WT = Path(".")
sys.path.insert(0, str(_WT / "tests" / "unit" / "aer"))

from aer_support import codes_of, edit, example_records, example_store, validate_example  # noqa: E402

from agentprov.aer.locators import LocatorResolver  # noqa: E402


@pytest.mark.parametrize("bad", [5, True, 1.5])
def test_scalar_artifacts_with_store_reports_not_crashes(bad):
    recs = edit(example_records(), "exec", artifacts=bad)
    without = codes_of(validate_example(recs, seal=False, with_store=False))
    assert "aer.schema.type" in without or without  # baseline: reported without a store
    report = validate_example(recs, seal=False, with_store=True)  # TypeError on current code
    assert codes_of(report)


@pytest.mark.parametrize("bad", [5, True, 1.5])
def test_resolver_tolerates_scalar_artifacts(bad):
    r = LocatorResolver({"artifacts": bad}, example_store())
    from agentprov.aer.locators import LocatorError
    with pytest.raises(LocatorError):
        r.artefact_bytes("a0")
